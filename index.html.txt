        function loadState() {
            try {
                const saved = localStorage.getItem('visitasa_local_db');
                if (saved) {
                    const parsed = JSON.parse(saved);
                    appState.nodes = parsed.nodes || DEFAULT_NODES;
                    appState.supervisors = parsed.supervisors || DEFAULT_SUPERVISORS;
                    appState.agencies = parsed.agencies || DEFAULT_AGENCIES;
                    appState.visitsMap = parsed.visitsMap || {};
                    appState.notesMap = parsed.notesMap || {};
                }
            } catch (e) { console.error("Error al cargar estado local", e); }
            populateSupervisorDropdowns();
        }

        function populateSupervisorDropdowns() {
            const nodeFilter = document.getElementById('headerGlobalNodeFilter');
            const globalFilter = document.getElementById('headerGlobalSupervisorFilter');
            
            const uniqueZones = [...new Set(appState.nodes.map(n => `${n.name} (${n.director})`))];

            if (nodeFilter) {
                let nodeHtml = `<option value="ALL">-- Todos los Nodos --</option>`;
                uniqueZones.forEach(z => nodeHtml += `<option value="${z}">${z}</option>`);
                nodeFilter.innerHTML = nodeHtml;
            }

            if (globalFilter) {
                let supHtml = `<option value="ALL">-- Todos los Jefes --</option>`;
                appState.supervisors.forEach(s => supHtml += `<option value="${s.id}">${s.name}</option>`);
                globalFilter.innerHTML = supHtml;
            }
        }

        function switchTab(tabId) {
            if (tabId === 'master' && !isMasterAuthenticated) {
                document.getElementById('modalMasterAuth').classList.remove('hidden');
                return;
            }
            appState.activeTab = tabId;
            document.querySelectorAll('.tab-content').forEach(sec => sec.classList.toggle('hidden', sec.id !== `tab-${tabId}`));
            renderActiveTab();
        }

        function renderActiveTab() {
            if (appState.activeTab === 'dashboard') renderDashboard();
            else if (appState.activeTab === 'matrix') renderMatrix();
        }

        function renderDashboard() {
            document.getElementById('kpiTotalAgencies').innerText = appState.agencies.length;
        }

        function renderMatrix() {
            const headerRow = document.getElementById('matrixHeaderRow');
            const tbody = document.getElementById('matrixTableBody');
            
            let headerHtml = `<th class="px-3 py-2 text-center">N°</th><th class="px-4 py-2">AGENCIA</th><th class="px-4 py-2">JEFE ZONA</th>`;
            for (let d = 1; d <= appState.daysInMonth; d++) headerHtml += `<th class="px-2 py-2 text-center">${d}</th>`;
            headerRow.innerHTML = headerHtml;

            let bodyHtml = "";
            appState.agencies.forEach((ag, idx) => {
                const sup = appState.supervisors.find(s => s.id === ag.supervisorId) || { name: 'Sin Asignar' };
                bodyHtml += `<tr><td class="px-3 py-2">${idx + 1}</td><td class="px-4 py-2 font-bold">${ag.name}</td><td class="px-4 py-2">${sup.name}</td>`;
                for (let d = 1; d <= appState.daysInMonth; d++) {
                    const key = getCellKey(ag.id, d);
                    const val = appState.visitsMap[key] || '';
                    bodyHtml += `<td class="px-2 py-2 text-center cursor-pointer border" onclick="toggleCell('${key}')">${val}</td>`;
                }
                bodyHtml += `</tr>`;
            });
            tbody.innerHTML = bodyHtml;
        }

        window.toggleCell = function(key) {
            const current = appState.visitsMap[key] || '';
            if (current === '') appState.visitsMap[key] = 'X';
            else if (current === 'X') appState.visitsMap[key] = '✓';
            else if (current === '✓') appState.visitsMap[key] = '!';
            else delete appState.visitsMap[key];
            saveState();
            renderMatrix();
        };

        document.addEventListener('DOMContentLoaded', () => {
            loadState();
            document.querySelectorAll('.tab-btn').forEach(btn => btn.addEventListener('click', () => switchTab(btn.dataset.tab)));
            document.getElementById('formMasterAuth')?.addEventListener('submit', (e) => {
                e.preventDefault();
                if (document.getElementById('authUsername').value === 'admin' && document.getElementById('authPassword').value === 'admin123') {
                    isMasterAuthenticated = true;
                    document.getElementById('modalMasterAuth').classList.add('hidden');
                    switchTab('master');
                } else {
                    document.getElementById('authErrorMsg').classList.remove('hidden');
                }
            });
            document.querySelectorAll('.btn-close-modal').forEach(b => b.addEventListener('click', () => document.getElementById('modalMasterAuth').classList.add('hidden')));
            renderActiveTab();
        });
    </script>
